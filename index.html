function renderTable(data) { const tbody = document.getElementById("table-body"); tbody.innerHTML = ""; if (data.length === 0) { tbody.innerHTML = `Belum ada data temuan.`; return; } data.forEach(item => { const progress = parseInt(item.Progress_Percent) || 0; const statusText = item.Status || (progress === 100 ? "Selesai" : "On Progress"); // Render Badge sesuai Nilai Status let statusBadge = `On Progress`; if (statusText === "Selesai" || progress === 100) { statusBadge = `Selesai`; } const nominalFormat = item.Jenis === 'Keuangan' ? 'Rp ' + Number(item.Nominal_Rp).toLocaleString('id-ID') : '-'; const disetorFormat = item.Jenis === 'Keuangan' ? `Rp ${Number(item.Total_Disetor).toLocaleString('id-ID')} (${progress}%)` : `${progress}%`; tbody.innerHTML += ` ${item.Sumber} ${item.Jenis} ${item.Judul_Temuan} ${nominalFormat} ${disetorFormat} ${statusBadge} `; }); } function renderMetrics(data) { document.getElementById("total-temuan").innerText = data.length; // Hitung berdasarkan status Selesai & On Progress document.getElementById("total-selesai").innerText = data.filter(d => d.Status === "Selesai" || parseInt(d.Progress_Percent) === 100).length; document.getElementById("total-proses").innerText = data.filter(d => d.Status === "On Progress" && parseInt(d.Progress_Percent) < 100).length; const totalUang = data.reduce((acc, curr) => acc + (Number(curr.Nominal_Rp) || 0), 0); document.getElementById("total-keuangan").innerText = "Rp " + totalUang.toLocaleString('id-ID'); }
